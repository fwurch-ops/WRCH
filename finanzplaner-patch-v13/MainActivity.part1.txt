    fun month(): MonthData {
        val k = monthKey(ym)
        if (!data.months.containsKey(k)) {
            val md = MonthData()
            val prev = data.months[monthKey(ym.minusMonths(1))]
            prev?.income?.filter { it.recurring }?.forEach { x ->
                val day = runCatching { LocalDate.parse(x.date).dayOfMonth }.getOrDefault(1)
                md.income += x.copy(id=id(), date=ym.atDay(minOf(day,ym.lengthOfMonth())).toString(), templateId=x.templateId ?: x.id)
            }
            data.months[k] = md
            store.save(data)
        }
        return data.months[k]!!
    }
    fun persist() { store.save(data); data = data.copy() }
    fun locked() = month().closed
    fun fixedValue(x:FixedItem):FixedOverride = x.overrides[monthKey(ym)] ?: FixedOverride(x.name,x.amount,x.startDate,x.frequency)

    fun occurs(x: FixedItem):Boolean { val v=fixedValue(x); val s=YearMonth.from(LocalDate.parse(v.startDate)); val diff=(ym.year-s.year)*12+ym.monthValue-s.monthValue; return when(v.frequency){Frequency.MONTHLY->diff>=0;Frequency.QUARTERLY->diff>=0&&diff%3==0;Frequency.YEARLY->diff>=0&&diff%12==0;Frequency.ONCE->ym==s} }
    fun fixedReserved()=data.fixed.sumOf{c->c.items.sumOf(::monthlyReserve)}
    fun incomeTotal()=month().income.sumOf{it.amount}
    fun allocated()=month().allocations.values.sum()
    fun spent(id:String)=month().expenses.filter{it.budgetId==id}.sumOf{it.amount}
    fun available()=incomeTotal()-fixedReserved()-allocated()
    fun savingBalance(id:String)=data.months.filterKeys{it<=monthKey(ym)}.values.flatMap{it.savingMoves}.filter{it.savingId==id}.sumOf{it.amount}
    fun dateForMonth(day:Int=LocalDate.now().dayOfMonth)=ym.atDay(minOf(day,ym.lengthOfMonth())).toString()
    fun actualBank():Double { val today=if(ym==YearMonth.now()) LocalDate.now() else ym.atEndOfMonth(); val inc=month().income.filter{LocalDate.parse(it.date)<=today}.sumOf{it.amount}; val fx=data.fixed.flatMap{it.items}.filter(::occurs).filter{ val v=fixedValue(it); val d=ym.atDay(minOf(LocalDate.parse(v.startDate).dayOfMonth,ym.lengthOfMonth())); d<=today }.sumOf{fixedValue(it).amount}; val ex=month().expenses.filter{it.source==Source.BANK&&LocalDate.parse(it.date)<=today}.sumOf{it.amount}; val tr=month().transfers.filter{LocalDate.parse(it.date)<=today}.sumOf{it.amount}; return month().openingBalance+inc-fx-ex-tr }
    fun cash()=month().transfers.sumOf{it.amount}-month().expenses.filter{it.source==Source.CASH}.sumOf{it.amount}
