                    opening={dialog=amountDialog("Startsaldo Girokonto","Kontostand am Monatsanfang",month().openingBalance){a->month().openingBalance=a;persist()}},

                    incomeDelete={x->month().income.remove(x);persist()},
                    close={val rest=data.budgets.sumOf{b->max(0.0,(month().allocations[b.id]?:0.0)-spent(b.id))};val sum=data.savings.sumOf{it.percent};if(sum!=100){message="Die Sparprozente ergeben $sum %. Sie müssen zusammen 100 % ergeben."}else{data.savings.forEach{s->val a=rest*s.percent/100.0;if(a>0)month().savingMoves+=SavingMove(savingId=s.id,amount=a,date=ym.atEndOfMonth().toString(),note="Monatsabschluss")};month().archive=ArchiveSnapshot(incomeTotal(),fixedReserved(),month().expenses.sumOf{it.amount},rest,"${data.budgets.size} Budgettöpfe · ${data.fixed.sumOf{it.items.size}} Fixkosten");month().closed=true;month().closedAt=System.currentTimeMillis();persist();message="Monat wurde abgeschlossen und archiviert."}})
            }
        }}
    }
    dialog?.let{DialogHost(it){dialog=null}}
    receiptDraft?.let{draft->ReceiptDialog(draft,data.budgets,{merchant,a,bid,src->month().expenses+=Expense(budgetId=bid,amount=a,date=dateForMonth(),source=src,note="$merchant · Kassenbon");persist();receiptDraft=null},{receiptDraft=null})}
    message?.let{AlertDialog(onDismissRequest={message=null},confirmButton={TextButton({message=null}){Text("OK")}},title={Text("WRCH Finanzplaner")},text={Text(it)})}
}
